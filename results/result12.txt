postgres=# WITH department_averages AS (
    SELECT
        department,
        AVG(salary) AS avg_salary
    FROM employees
    GROUP BY department
)
SELECT
    e.name,
    e.salary,
    e.department,
    da.avg_salary
FROM employees e
JOIN department_averages da
    ON e.department = da.department
WHERE e.salary > da.avg_salary;
